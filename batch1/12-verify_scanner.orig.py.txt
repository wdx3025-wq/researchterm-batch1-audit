# -*- coding: utf-8 -*-
"""
P35 + P39 + P41 的实现：扫描器「单一来源」自动一致性检查。

检查项：
  1) 文档 §7.5.1 里的代码块，与 tmp/scanner.py 的**可注入主体**逐字节一致
     （抽取规则由 `scan_extract.injectable()` 提供，构建器和本检查器共用同一份，不再各写一套）；
  2) 会话里没有任何手抄的第二份实现（模板只留占位符）；
  3) 注入点唯一（P39）：模板 / 中间 Markdown / HTML 三个层次各只有一份可执行代码。

用法：
  python tmp/verify_scanner.py           # 正常检查（应全绿）
  python tmp/verify_scanner.py --mutate  # 故意改文档里的代码制造差异，验证检查器能发现
"""
import pathlib
import re
import sys

try:
    sys.stdout.reconfigure(encoding="utf-8")
except Exception:
    pass

HERE = pathlib.Path(__file__).resolve().parent
ROOT = HERE.parent
sys.path.insert(0, str(HERE))
from scan_extract import injectable

MD = (ROOT / "docs" / "architecture" / "plan-v6.2-terms.md").read_text(encoding="utf-8")
SCANNER = (HERE / "scanner.py").read_text(encoding="utf-8")
TPL = (HERE / "doc_v62.tpl.md").read_text(encoding="utf-8")
SITE = ROOT / "site"

BODY = injectable(SCANNER)
FAIL = 0

# 含 scan() 定义的页面**完整集合**。三张都是同一份 BODY 的一次展示，不是第二份实现：
#   ch07.html          设计文档第 7 章 §7.5.1 正文（单一来源本体）
#   index.html         合订本，同一份代码在完整版里的第二次展示（P39 允许的"两张视图"）
#   review-scanner.html 复核分册「被审计对象全文」，给外部审阅者看的第三处展示
# 约束不变：**每页至多 1 份**。新增页面必须改这里并说明理由，不允许"多出来就放过"。
EXPECTED_SCAN_PAGES = ["ch07.html", "index.html", "review-scanner.html"]


def html_pages_ok(counts):
    """页面集合必须与 EXPECTED_SCAN_PAGES **精确相等**，且每页至多 1 份。

    返回 (是否通过, 说明)。拆成纯函数是为了能对判定本身做负例（见第 4 节），
    避免"检查器自己写错却永远为真"。
    """
    pages = sorted(counts)
    if pages != EXPECTED_SCAN_PAGES:
        missing = [p for p in EXPECTED_SCAN_PAGES if p not in pages]
        extra = [p for p in pages if p not in EXPECTED_SCAN_PAGES]
        return False, f"页面集合不符：缺 {missing}，多 {extra}，实际 {pages}"
    dup = {k: v for k, v in counts.items() if v != 1}
    if dup:
        return False, f"单页出现多份 scan()：{dup}"
    return True, f"集合精确匹配 {pages}，每页恰 1 份"


def sec(t):
    print("\n" + "=" * 70)
    print(t)
    print("=" * 70)


def check(name, ok):
    global FAIL
    print(f"    {'✓' if ok else '✗'} {name}")
    if not ok:
        FAIL = 1


def scanner_blocks(text):
    """取出 §7.5.1 里「含 scan() 定义」的 python 代码块。"""
    m = re.search(r"^### 7\.5\.1 .*?(?=^### 7\.6)", text, re.S | re.M)
    if not m:
        return None, []
    out = []
    for f in re.findall(r"```python\n(.*?)```", m.group(0), re.S):
        if re.search(r"^def scan\(conn, mark\):", f, re.M):
            out.append(f)
    return m, out


# 1. 源文件构成
sec("1. scanner.py 的可注入主体")
print(f"  总行数 {len(SCANNER.splitlines())} 行；可注入主体 {len(BODY.splitlines())} 行")
print(f"  主体首行：{BODY.splitlines()[0]}")
check("主体里有 import re", "import re" in BODY)
check("主体里有 def scan(", "def scan(conn, mark):" in BODY)
check("主体里有 def _q(", "def _q(ident: str) -> str:" in BODY)
check("模块 docstring 未被注入", "这份实现是唯一权威版本" not in BODY)
# P42：文档里展示的必须是"两层判据"真的在用，而不是只在说明文字里宣称
check("主体里有类型层判据（types.get != virtual）",
      'types.get(n) != "virtual"' in BODY)
check("主体里有 PRAGMA table_list 调用", "PRAGMA table_list" in BODY)
check("主体里有版本门（_table_list_usable）", "_table_list_usable()" in BODY)
check("主体里没有 catch-all 式降级（except sqlite3.OperationalError）",
      "except sqlite3.OperationalError" not in BODY)

# 2. 逐字节比对
sec("2. 文档 §7.5.1 代码块 与 scanner.py 逐字节比对")
m, blocks = scanner_blocks(MD)
if m is None:
    print("  ✗ 找不到 §7.5.1")
    FAIL = 1
else:
    print(f"  §7.5.1 里含 scan() 定义的代码块数：{len(blocks)}（应为 1）")
    check("代码块数量为 1", len(blocks) == 1)
    if blocks:
        doc_block = blocks[0].strip()
        src_block = BODY.strip()
        same = doc_block == src_block
        print(f"  文档块 {len(doc_block)} 字符 / 源文件主体 {len(src_block)} 字符")
        if not same:
            import difflib
            diff = list(difflib.unified_diff(src_block.splitlines(), doc_block.splitlines(),
                                             "scanner.py", "doc", lineterm="", n=1))
            print("  差异（前 20 行）：")
            for ln in diff[:20]:
                print("   ", ln)
        check("文档代码 = scanner.py（逐字节）", same)

# 3. 注入唯一性（P39）
sec("3. 注入唯一性：模板 / 中间 Markdown / HTML 各自只有一份可执行代码")
tpl_line = len(re.findall(r"^[ \t]*\{\{PY:scanner_scan\}\}[ \t]*$", TPL, re.M))
tpl_inline = len(re.findall(r"\{\{PY:scanner_scan\}\}", TPL)) - tpl_line
tpl_code = len(re.findall(r"^def scan\(conn, mark\):", TPL, re.M))
print(f"  模板：独占行占位符 {tpl_line} 处（应为 1）、行内提及 {tpl_inline} 处、"
      f"手写源码 {tpl_code} 处（应为 0）")
md_scan = len(re.findall(r"^def scan\(conn, mark\):", MD, re.M))
md_q = len(re.findall(r"^def _q\(ident: str\) -> str:", MD, re.M))
unresolved = re.findall(r"^[ \t]*\{\{(?:SQL|PY):\w+\}\}[ \t]*$", MD, re.M)
print(f"  Markdown：def scan( {md_scan} 次、def _q( {md_q} 次、未解析独占行占位符 {len(unresolved)} 处")
html_counts = {}
if SITE.is_dir():
    for p in sorted(SITE.glob("*.html")):
        txt = p.read_text(encoding="utf-8", errors="ignore")
        n = len(re.findall(r"^def scan\(conn, mark\):", txt, re.M))
        if n:
            html_counts[p.name] = n
    print(f"  HTML：含 scan() 定义的页面 {html_counts if html_counts else '{}'}")
    pages = sorted(html_counts)
    total_html = sum(html_counts.values())
else:
    print("  HTML：site/ 不存在，跳过（先跑 md2html.py）")
    pages, total_html = [], 0
check("模板里独占行占位符恰好 1 处", tpl_line == 1)
check("模板里没有手写源码（只认占位符）", tpl_code == 0)
check("Markdown 里 def scan( 恰好 1 次", md_scan == 1)
check("Markdown 里 def _q( 恰好 1 次", md_q == 1)
check("Markdown 里没有未解析的独占行占位符", not unresolved)
if html_counts:
    ok_pages, why_pages = html_pages_ok(html_counts)
    print(f"  集合判定：{why_pages}")
    check("HTML 每页至多 1 份 scan()", all(v == 1 for v in html_counts.values()))
    check("HTML 含 scan() 的页面集合与 EXPECTED_SCAN_PAGES 精确相等", ok_pages)

# 4. 负例：判定函数本身必须能拒绝错误集合（不依赖 site/ 现状）
sec("4. 负例验证：判定函数能拒绝错误集合（检错能力不依赖目录恰好正确）")
NEG = [
    ("缺 review-scanner.html（回到两张视图）",
     {"ch07.html": 1, "index.html": 1}, False),
    ("缺合订本 index.html",
     {"ch07.html": 1, "review-scanner.html": 1}, False),
    ("多出一张陌生页面",
     {"ch07.html": 1, "index.html": 1, "review-scanner.html": 1, "other.html": 1}, False),
    ("index.html 出现 2 份 scan()（合订本被注入两次）",
     {"ch07.html": 1, "index.html": 2, "review-scanner.html": 1}, False),
    ("review-scanner.html 出现 2 份 scan()",
     {"ch07.html": 1, "index.html": 1, "review-scanner.html": 2}, False),
    ("空集合（site/ 里一份都没有）", {}, False),
    ("完全正确（三页各 1 份）",
     {"ch07.html": 1, "index.html": 1, "review-scanner.html": 1}, True),
]
for desc, counts, want in NEG:
    got, why = html_pages_ok(counts)
    mark = "✓" if got == want else "✗"
    print(f"    {mark} {desc}")
    print(f"        → {'通过' if got else '拒绝'}（期望{'通过' if want else '拒绝'}）{why}")
    check(f"负例：{desc}", got == want)

# 5. 负例：篡改文档里的判据，检查器必须发现
sec("5. 负例验证：篡改文档正文，检查器必须能发现")
if "--mutate" in sys.argv:
    polluted = MD.replace('types.get(n) != "virtual"', 'types.get(n) != "virtual_x"')
    _, b2 = scanner_blocks(polluted)
    caught = bool(b2) and b2[0].strip() != BODY.strip()
    check("篡改类型层判据（virtual → virtual_x）后能报出不一致", caught)
else:
    print("  （跳过：加 --mutate 运行此负例）")

# 6. P39 对照实验 + 重复检出
sec("6. P39 对照实验：旧「全局替换」会造出多份，新「独占行替换」只造一份")
naive = re.sub(r"\{\{PY:(\w+)\}\}", lambda _m: BODY, TPL)
strict = re.sub(r"^[ \t]*\{\{PY:(\w+)\}\}[ \t]*$", lambda _m: BODY, TPL, flags=re.M)
n_naive = len(re.findall(r"^def scan\(conn, mark\):", naive, re.M))
n_strict = len(re.findall(r"^def scan\(conn, mark\):", strict, re.M))
print(f"  旧规则（全局 re.sub）：产出 {n_naive} 份 def scan( —— 这就是 v6.2.5 线上那份的来历")
print(f"  新规则（只认独占行）：产出 {n_strict} 份")
check("旧规则确实会产出多份（bug 可复现）", n_naive > 1)
check("新规则只产出 1 份", n_strict == 1)

if m:
    dup = MD[:m.end(0)] + "\n```python\n" + BODY + "\n```\n" + MD[m.end(0):]
    _, b3 = scanner_blocks(dup)
    print(f"  人为把代码块再复制一份后，§7.5.1 里可解析出 scan() 的块数 = {len(b3)}")
    check("重复代码块能被检出（应数为 2）", len(b3) == 2)

print("\n" + "=" * 70)
print("总判定：" + ("通过（单一来源成立，注入唯一）" if not FAIL else "存在失败项"))
print("=" * 70)
sys.exit(FAIL)
