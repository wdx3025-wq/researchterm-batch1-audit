"""
P35 + P39 + P41 的实现：扫描器「单一来源」自动一致性检查。

检查项：
  1) 文档 §7.5.1 里的代码块，与 tmp/scanner.py 的**可注入主体**逐字节一致
     （抽取规则由 `scan_extract.injectable()` 提供，构建器和本检查器共用同一份，不再各写一套）；
  2) 会话里没有任何手抄的第二份实现（模板只留占位符）；
  3) 注入点唯一（P39）：模板 / 中间 Markdown / HTML 三个层次各只有一份可执行代码。

用法：
  python tmp/verify_scanner.py           # 正常检查（应全绿）
  python tmp/verify_scanner.py --mutate  # 故意改文档里的代码制造差异，验证检查器能发现
"""
import pathlib
import re
import shutil
import sys
import tempfile

try:
    sys.stdout.reconfigure(encoding="utf-8")
except Exception:
    pass

HERE = pathlib.Path(__file__).resolve().parent
ROOT = HERE.parent
sys.path.insert(0, str(HERE))
from scan_extract import injectable

MD = (ROOT / "docs" / "architecture" / "plan-v6.2-terms.md").read_text(encoding="utf-8")
SCANNER = (HERE / "scanner.py").read_text(encoding="utf-8")
TPL = (HERE / "doc_v62.tpl.md").read_text(encoding="utf-8")
SITE = ROOT / "site"

BODY = injectable(SCANNER)
FAIL = 0

# 含 scan() 定义的页面**完整集合**。三张都是同一份 BODY 的一次展示，不是第二份实现：
#   ch07.html          设计文档第 7 章 §7.5.1 正文（单一来源本体）
#   index.html         合订本，同一份代码在完整版里的第二次展示（P39 允许的"两张视图"）
#   review-scanner.html 复核分册「被审计对象全文」，给外部审阅者看的第三处展示
# 约束不变：**每页至多 1 份**。新增页面必须改这里并说明理由，不允许"多出来就放过"。
EXPECTED_SCAN_PAGES = ["ch07.html", "index.html", "review-scanner.html"]

# 原因码：机读，测试断言 code 而不只是 ok
REASON_SITE_MISSING = "SITE_MISSING"     # site/ 不存在 —— 构建产物缺失
REASON_ZERO_HIT     = "ZERO_HIT"         # 目录在，但没有任何页面出现 def scan(
REASON_DUPLICATE    = "DUPLICATE"        # 某页出现多份 scan()
REASON_SET_MISMATCH = "SET_MISMATCH"     # 页面集合与 EXPECTED_SCAN_PAGES 不等
REASON_OK           = "OK"


def html_pages_ok(counts):
    """页面集合必须与 EXPECTED_SCAN_PAGES **精确相等**，且每页至多 1 份。

    返回 (是否通过, 说明)。拆成纯函数是为了能对判定本身做负例（见第 4 节），
    避免"检查器自己写错却永远为真"。
    """
    pages = sorted(counts)
    if pages != EXPECTED_SCAN_PAGES:
        missing = [p for p in EXPECTED_SCAN_PAGES if p not in pages]
        extra = [p for p in pages if p not in EXPECTED_SCAN_PAGES]
        return False, f"页面集合不符：缺 {missing}，多 {extra}，实际 {pages}"
    dup = {k: v for k, v in counts.items() if v != 1}
    if dup:
        return False, f"单页出现多份 scan()：{dup}"
    return True, f"集合精确匹配 {pages}，每页恰 1 份"


def _page_diff(counts):
    """(排序后的实际页面, 缺页清单, 多页清单)。只用于生成诊断文本。"""
    pages = sorted(counts)
    missing = [p for p in EXPECTED_SCAN_PAGES if p not in pages]
    extra = [p for p in pages if p not in EXPECTED_SCAN_PAGES]
    return pages, missing, extra


def _set_diag(counts):
    """集合诊断串（**只进 why 文本，不参与判定**）。集合本身正常时明说"无差异"。"""
    _, missing, extra = _page_diff(counts)
    if not missing and not extra:
        return "集合诊断：无差异"
    return f"集合诊断：缺 {missing}，多 {extra}"


def scan_page_counts(site_dir):
    """取数：{页面名: 该页 def scan( 出现次数}；目录不存在返回 None。

    None 与 {} 严格区分：前者是"构建产物缺失"，后者是"产物在但零命中"，
    两类故障的排查方向完全不同，不能合并。
    """
    p = pathlib.Path(site_dir)
    if not p.is_dir():
        return None
    counts = {}
    for f in sorted(p.glob("*.html")):
        txt = f.read_text(encoding="utf-8", errors="ignore")
        n = len(re.findall(r"^def scan\(conn, mark\):", txt, re.M))
        if n:
            counts[f.name] = n
    return counts


def judge_site_pages(site_dir):
    """页面集合的**唯一**判定入口（主流程与负例都调它）。

    返回 (ok: bool, code: str, why: str, counts)。
    判定顺序（已定稿，短路，只报第一个命中的原因码）：
        SITE_MISSING → ZERO_HIT → DUPLICATE → SET_MISMATCH → OK
    counts 永远是**完整**的取数结果（不因短路而截断），供诊断使用；
    目录不存在时它就是 scan_page_counts() 的返回值 None。
    """
    counts = scan_page_counts(site_dir)
    if counts is None:
        return (False, REASON_SITE_MISSING,
                f"目录不存在：{site_dir}（先跑 md2html.py 生成 site/）", counts)
    if not counts:
        return (False, REASON_ZERO_HIT,
                f"目录内没有任何页面出现 def scan(（应为 {EXPECTED_SCAN_PAGES}）", counts)
    # 判定本身复用 html_pages_ok()，不复制一套集合逻辑
    ok, why = html_pages_ok(counts)
    if ok:
        return True, REASON_OK, why, counts
    # 归因专用：判定结果已由 html_pages_ok() 给出，这里再查一次 dup
    # 只是为了把"失败"分类到 DUPLICATE / SET_MISMATCH，**不参与判定**。
    dup = {k: counts[k] for k in sorted(counts) if counts[k] != 1}
    if dup:
        return False, REASON_DUPLICATE, f"某页出现多份 scan()：{dup}；{_set_diag(counts)}", counts
    _, missing, extra = _page_diff(counts)
    return False, REASON_SET_MISMATCH, f"缺 {missing}，多 {extra}", counts


def sec(t):
    print("\n" + "=" * 70)
    print(t)
    print("=" * 70)


def check(name, ok):
    global FAIL
    print(f"    {'✓' if ok else '✗'} {name}")
    if not ok:
        FAIL = 1


def scanner_blocks(text):
    """取出 §7.5.1 里「含 scan() 定义」的 python 代码块。"""
    m = re.search(r"^### 7\.5\.1 .*?(?=^### 7\.6)", text, re.S | re.M)
    if not m:
        return None, []
    out = []
    for f in re.findall(r"```python\n(.*?)```", m.group(0), re.S):
        if re.search(r"^def scan\(conn, mark\):", f, re.M):
            out.append(f)
    return m, out

# 3. 注入唯一性（P39）
sec("3. 注入唯一性：模板 / 中间 Markdown / HTML 各自只有一份可执行代码")

tpl_inline = len(re.findall(r"\{\{PY:scanner_scan\}\}", TPL)) - tpl_line
tpl_code = len(re.findall(r"^def scan\(conn, mark\):", TPL, re.M))

# 4. 负例：判定函数本身必须能拒绝错误集合（不依赖 site/ 现状）
sec("4. 负例验证：判定函数能拒绝错误集合（检错能力不依赖目录恰好正确）")

# 4b. 流程级用例：**走正式入口 judge_site_pages()**（临时目录构造现场，不动 site/）
sec("4b. 流程级用例：judge_site_pages() 逐条断原因码（6 条，临时目录）")
STUB = "def scan(conn, mark):\n    return {}\n"


def _make_site(root, spec):
    """按 {页面名: scan() 份数} 造临时站点；份数 0 表示该页不含 scan()。"""
    root.mkdir(parents=True, exist_ok=True)
    for name, times in spec.items():
        body = STUB * times if times else "<html>no scanner</html>\n"
        (root / name).write_text(body, encoding="utf-8")
    return root


FLOW_CASES = [
    ("1",  "site 目录不存在",         None,
     False, REASON_SITE_MISSING),
    ("2",  "存在但零命中",            {"zzz.html": 0},
     False, REASON_ZERO_HIT),
    ("3",  "缺复核分册（ch07+index）", {"ch07.html": 1, "index.html": 1},
     False, REASON_SET_MISMATCH),
    ("4",  "单页 2 份，集合正常",
     {"ch07.html": 1, "index.html": 2, "review-scanner.html": 1},
     False, REASON_DUPLICATE),
    ("4b", "单页 2 份 且 集合异常",
     {"ch07.html": 1, "index.html": 2, "zzz.html": 1},
     False, REASON_DUPLICATE),
    ("5",  "正确三页",
     {"ch07.html": 1, "index.html": 1, "review-scanner.html": 1},
     True, REASON_OK),
]

tmp_root = pathlib.Path(tempfile.mkdtemp(prefix="batch1_ta_"))
try:
    for cid, desc, spec, want_ok, want_code in FLOW_CASES:
        site_dir = (tmp_root / "nonexistent" if spec is None
                    else _make_site(tmp_root / f"case_{cid}", spec))
        ok, code, why, counts = judge_site_pages(site_dir)
        good = (ok == want_ok and code == want_code)
        print(f"    {'✓' if good else '✗'} 用例 {cid}｜{desc} → ok={ok} [{code}] {why}")
        print(f"        counts={counts}（期望 ok={want_ok} code={want_code}）")
        check(f"流程用例 {cid}：ok={want_ok} 且 code={want_code}", good)
        if cid == "4":
            check("流程用例 4：集合正常时 why 明说「无差异」", "无差异" in why)
        if cid == "4b":
            check("流程用例 4b：why 同时含缺页与多页字样", "缺" in why and "多" in why)
            check("流程用例 4b：counts 为完整字典（未被短路截断）",
                  counts == {"ch07.html": 1, "index.html": 2, "zzz.html": 1})
finally:
    shutil.rmtree(tmp_root, ignore_errors=True)

# 6. P39 对照实验 + 重复检出
sec("6. P39 对照实验：旧「全局替换」会造出多份，新「独占行替换」只造一份")
naive = re.sub(r"\{\{PY:(\w+)\}\}", lambda _m: BODY, TPL)
strict = re.sub(r"^[ \t]*\{\{PY:(\w+)\}\}[ \t]*$", lambda _m: BODY, TPL, flags=re.M)
n_naive = len(re.findall(r"^def scan\(conn, mark\):", naive, re.M))
n_strict = len(re.findall(r"^def scan\(conn, mark\):", strict, re.M))
print(f"  旧规则（全局 re.sub）：产出 {n_naive} 份 def scan( —— 这就是 v6.2.5 线上那份的来历")
print(f"  新规则（只认独占行）：产出 {n_strict} 份")
check("旧规则确实会产出多份（bug 可复现）", n_naive > 1)
check("新规则只产出 1 份", n_strict == 1)

if m:
    dup = MD[:m.end(0)] + "\n```python\n" + BODY + "\n```\n" + MD[m.end(0):]
    _, b3 = scanner_blocks(dup)
    print(f"  人为把代码块再复制一份后，§7.5.1 里可解析出 scan() 的块数 = {len(b3)}")
    check("重复代码块能被检出（应数为 2）", len(b3) == 2)

print("\n" + "=" * 70)
